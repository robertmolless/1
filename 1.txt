// ===== Анимация =====
let frame = 0;
let moodTimer = 0;

function animate() {
  frame++;

  // Хвост
  state.tailAngle += 0.04 * state.tailDir;
  if (Math.abs(state.tailAngle) > 0.6) state.tailDir *= -1;

  // Моргание
  state.blinkTimer--;
  if (state.blinkTimer < 0) {
    state.eyeClose = 4;
    state.blinkTimer = 120 + Math.random() * 100;
  }
  if (state.eyeClose > 0) state.eyeClose--;

  // Сброс настроения
  moodTimer--;
  if (moodTimer < 0) {
    state.mood = 'normal';
  }

  drawCat(state.mood);

  // Мурлыканье визуальное
  if (state.purring && frame % 40 === 0) {
    spawnPurr();
  }

  requestAnimationFrame(animate);
}

animate();

// ===== Звуки через Web Audio =====
const AudioCtx = window.AudioContext || window.webkitAudioContext;
let audioCtx = null;

function getAudio() {
  if (!audioCtx) audioCtx = new AudioCtx();
  return audioCtx;
}

function playMeow() {
  try {
    const ac = getAudio();
    const osc = ac.createOscillator();
    const gain = ac.createGain();
    osc.connect(gain); gain.connect(ac.destination);
    osc.type = 'sine';
    osc.frequency.setValueAtTime(600, ac.currentTime);
    osc.frequency.exponentialRampToValueAtTime(900, ac.currentTime + 0.1);
    osc.frequency.exponentialRampToValueAtTime(500, ac.currentTime + 0.35);
    gain.gain.setValueAtTime(0.3, ac.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, ac.currentTime + 0.4);
    osc.start(); osc.stop(ac.currentTime + 0.4);
  } catch(e) {}
}

function playPurr() {
  try {
    const ac = getAudio();
    const osc = ac.createOscillator();
    const gain = ac.createGain();
    const lfo = ac.createOscillator();
    const lfoGain = ac.createGain();
    lfo.connect(lfoGain); lfoGain.connect(gain.gain);
    osc.connect(gain); gain.connect(ac.destination);
    osc.type = 'sawtooth';
    osc.frequency.value = 55;
    lfo.frequency.value = 25;
    lfoGain.gain.value = 0.15;
    gain.gain.value = 0.12;
    osc.start(); lfo.start();
    osc.stop(ac.currentTime + 1.5);
    lfo.stop(ac.currentTime + 1.5);
    gain.gain.exponentialRampToValueAtTime(0.001, ac.currentTime + 1.5);
  } catch(e) {}
}

function playEat() {
  try {
    const ac = getAudio();
    [0, 0.15, 0.3].forEach(t => {
      const osc = ac.createOscillator();
      const gain = ac.createGain();
      osc.connect(gain); gain.connect(ac.destination);
      osc.type = 'sine';
      osc.frequency.value = 300 + Math.random() * 100;
      gain.gain.setValueAtTime(0.15, ac.currentTime + t);
      gain.gain.exponentialRampToValueAtTime(0.001, ac.currentTime + t + 0.1);
      osc.start(ac.currentTime + t);
      osc.stop(ac.currentTime + t + 0.12);
    });
  } catch(e) {}
}

// ===== Обновление UI =====
function clamp(v) { return Math.max(0, Math.min(100, v)); }

function updateStats() {
  document.getElementById('barHunger').style.width = state.hunger + '%';
  document.getElementById('barHappy').style.width = state.happy + '%';
  document.getElementById('barEnergy').style.width = state.energy + '%';

  let moodText = '';
  if (state.hunger < 20) moodText = 'Рыжик голоден 😿';
  else if (state.energy < 20) moodText = 'Рыжик хочет спать 😴';
  else if (state.happy > 70) moodText = 'Рыжик очень доволен 😸';
  else if (state.happy < 30) moodText = 'Рыжик скучает... 🙁';
  else moodText = 'Рыжик в порядке 😺';
  document.getElementById('mood').textContent = moodText;
}

// ===== Действия =====
function feedCat() {
  if (state.hunger >= 95) { showBubble('Я сыт!'); return; }
  state.hunger = clamp(state.hunger + 25);
  state.happy = clamp(state.happy + 5);
  state.mood = 'eating';
  moodTimer = 80;
  state.purring = false;
  playEat();
  spawnEffect('🍗', 80, 60);
  spawnEffect('😋', 30, 40);
  showBubble('Мням-мням!');
  updateStats();
}

function petCatBtn() {
  doPet(60, 100);
}

function petCat(e) {
  const rect = e.currentTarget.getBoundingClientRect();
  const x = e.clientX - rect.left;
  const y = e.clientY - rect.top;
  doPet(x, y);
}

function doPet(x, y) {
  state.happy = clamp(state.happy + 12);
  state.mood = 'happy';
  moodTimer = 90;
  state.purring = true;
  playPurr();
  const emojis = ['💛','✨','🧡','💫'];
  spawnEffect(emojis[Math.floor(Math.random()*emojis.length)], x - 15, y - 20);
  const phrases = ['Муррр~', 'Ещё!', 'Хорошо~', 'Пуррр...'];
  showBubble(phrases[Math.floor(Math.random()*phrases.length)]);
  setTimeout(() => { state.purring = false; }, 3000);
  updateStats();
}

function playCat() {
  if (state.energy < 15) { showBubble('Устал...'); return; }
  state.happy = clamp(state.happy + 20);
  state.energy = clamp(state.energy - 15);
  state.mood = 'playing';
  moodTimer = 100;
  state.purring = false;
  playMeow();
  spawnEffect('🧶', 20, 80);
  spawnEffect('⚡', 90, 50);
  showBubble('МЯУ! Играем!');
  updateStats();
}

function sleepCat() {
  if (state.energy >= 95) { showBubble('Не хочу спать!'); return; }
  state.energy = clamp(state.energy + 35);
  state.mood = 'sleep';
  moodTimer = 150;
  state.purring = false;
  spawnEffect('💤', 70, 30);
  showBubble('Zzz...');
  updateStats();
}

// ===== Деградация со временем =====
setInterval(() => {
  state.hunger = clamp(state.hunger - 2);
  state.happy = clamp(state.happy - 1);
  state.energy = clamp(state.energy - 0.5);
  updateStats();
}, 10000);
